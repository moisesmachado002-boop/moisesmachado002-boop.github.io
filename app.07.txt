  function deleteCard(id) {
    const card = state.cards.find(c => c.id === id);
    if (!card) return;
    const purchaseCount = state.purchases.filter(p => p.cardId === id).length;
    const msg = purchaseCount ? `Excluir o cartão "${card.name}" e também ${plural(purchaseCount, 'compra')} vinculada?` : `Excluir o cartão "${card.name}"?`;
    if (!confirm(msg)) return;
    state.cards = state.cards.filter(c => c.id !== id);
    state.purchases = state.purchases.filter(p => p.cardId !== id);
    saveState(); renderAll(); toast('Cartão excluído.');
  }

  function deletePurchase(id) {
    const p = state.purchases.find(x => x.id === id);
    if (!p || !confirm(`Excluir a compra "${p.description}" e todas as parcelas?`)) return;
    state.purchases = state.purchases.filter(x => x.id !== id);
    saveState(); renderAll(); toast('Compra excluída.');
  }

  function updateInstallmentPreview() {
    const form = els.purchaseForm;
    const amount = parseMoney(form.elements.amount.value);
    const installments = Math.max(1, Number(form.elements.installments.value) || 1);
    const card = state.cards.find(c => c.id === form.elements.cardId.value);
    if (!(amount > 0)) {
      els.installmentPreview.textContent = 'Informe o valor para visualizar as parcelas.';
      return;
    }
    const parts = installmentAmounts(amount, installments);
    const firstMonth = card && form.elements.date.value ? firstInvoiceMonth(form.elements.date.value, card.closingDay) : null;
    els.installmentPreview.textContent = `${installments}x de aproximadamente ${money(parts[0])}${firstMonth ? ` · primeira fatura: ${formatMonth(firstMonth)}` : ''}.`;
  }

  function renderBills() {
    const month = els.billsMonth.value || currentMonth();
    const items = state.bills.filter(b => b.active !== false).sort((a, b) => a.dueDay - b.dueDay);
    if (!items.length) return setEmpty(els.billsList, 'Nenhuma conta fixa cadastrada.');
    els.billsList.classList.remove('empty-state');
    els.billsList.innerHTML = items.map(bill => {
      const paid = isBillPaid(bill.id, month);
      return `<article class="list-item">
        <div class="item-main">
          <div class="item-title"><span>${escapeHtml(bill.description)}</span><span class="badge">${paid ? 'Paga' : 'Pendente'}</span></div>
          <div class="item-meta">${escapeHtml(bill.category)}${bill.paymentMethod ? ` · ${escapeHtml(bill.paymentMethod)}` : ''} · vence dia ${bill.dueDay} · ${formatMonth(month)}</div>
          <div class="item-actions">
            ${paid ? `<button class="mini-btn" data-unpay-bill="${bill.id}">Desfazer pagamento</button>` : `<button class="mini-btn" data-pay-bill="${bill.id}">Marcar como paga</button>`}
            <button class="mini-btn danger-text" data-delete-bill="${bill.id}">Excluir</button>
          </div>
        </div>
        <div class="amount ${paid ? '' : 'expense'}">${money(bill.amount)}</div>
      </article>`;
    }).join('');
    $$('[data-pay-bill]', els.billsList).forEach(btn => btn.addEventListener('click', () => payBill(btn.dataset.payBill, month)));
    $$('[data-unpay-bill]', els.billsList).forEach(btn => btn.addEventListener('click', () => unpayBill(btn.dataset.unpayBill, month)));
    $$('[data-delete-bill]', els.billsList).forEach(btn => btn.addEventListener('click', () => deleteBill(btn.dataset.deleteBill)));
  }
