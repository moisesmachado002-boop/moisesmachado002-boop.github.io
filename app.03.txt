  async function activateUser(user) {
    if (!user) return;
    currentUser = user;
    activeStorageKey = userStorageKey(user.id);
    els.cloudUserEmail.textContent = user.email || 'Conta conectada';
    setSyncStatus('Sincronizando...', 'syncing');
    setAuthMessage('Carregando seus dados...');

    const localUserState = readStoredState(activeStorageKey);
    const legacyState = loadLegacyState();

    let cloudRow = null;
    let cloudError = null;
    try {
      const { data, error } = await supabaseClient
        .from('finance_state')
        .select('data, updated_at')
        .eq('user_id', user.id)
        .maybeSingle();
      cloudRow = data;
      cloudError = error;
    } catch (error) {
      cloudError = error;
    }

    if (cloudError) {
      console.error('Erro ao carregar nuvem:', cloudError);
      const fallback = localUserState || defaultState();
      state = normalizeState(fallback);
      persistLocalState();
      cloudDirty = false;
      showApp();
      populateOpeningBalancesForm();
      renderAll();
      setSyncStatus(navigator.onLine ? 'Nuvem indisponível' : 'Sem internet', navigator.onLine ? 'error' : 'offline');
      subscribeRealtime();
      return;
    }

    const remoteState = cloudRow?.data ? normalizeState(cloudRow.data) : null;
    if (remoteState) {
      if (localUserState && stateTimestamp(localUserState) > stateTimestamp(remoteState)) {
        state = normalizeState(localUserState);
        cloudDirty = true;
        await pushCloudState(true);
      } else {
        state = remoteState;
        cloudDirty = false;
        persistLocalState();
      }
    } else {
      let seed = localUserState;
      if (!hasMeaningfulData(seed) && hasMeaningfulData(legacyState)) {
        const migratedFor = localStorage.getItem('meu_financeiro_legacy_migrated_for');
        if (!migratedFor || migratedFor === user.id) {
          const useLegacy = confirm('Encontramos dados da versão anterior neste aparelho. Deseja enviar esses dados para sua conta na nuvem?');
          if (useLegacy) {
            seed = legacyState;
            localStorage.setItem('meu_financeiro_legacy_migrated_for', user.id);
          }
        }
      }
      state = normalizeState(seed || defaultState());
      if (!state._meta.updatedAt) state._meta.updatedAt = new Date().toISOString();
      persistLocalState();
      cloudDirty = true;
      await pushCloudState(true);
    }

    showApp();
    populateOpeningBalancesForm();
    renderAll();
    subscribeRealtime();
    setSyncStatus('Sincronizado', 'synced');
  }

  function deactivateUser() {
    currentUser = null;
    activeStorageKey = null;
    state = defaultState();
    cloudDirty = false;
    if (realtimeChannel && supabaseClient) supabaseClient.removeChannel(realtimeChannel);
    realtimeChannel = null;
    showAuthGate();
  }

  async function signOut() {
    if (!supabaseClient) return;
    await pushCloudState();
    await supabaseClient.auth.signOut();
    deactivateUser();
  }

  function showApp() {
    authReady = true;
    els.authGate.classList.add('hidden');
    els.appShell.classList.remove('auth-hidden');
  }

  function showAuthGate() {
    authReady = false;
    els.appShell.classList.add('auth-hidden');
    els.authGate.classList.remove('hidden');
    setAuthMessage('Entre com sua conta para acessar e sincronizar os dados.');
  }

  function setAuthBusy(busy, message = '') {
    els.authForm.querySelectorAll('button, input').forEach(el => el.disabled = busy);
    if (message) setAuthMessage(message);
  }

  function setAuthMessage(message, kind = '') {
    els.authMessage.textContent = message;
    els.authMessage.className = `auth-message${kind ? ` ${kind}` : ''}`;
  }

  function friendlyAuthError(error) {
    const text = String(error?.message || error || 'Erro desconhecido');
    const lower = text.toLowerCase();
    if (lower.includes('invalid login credentials')) return 'E-mail ou senha incorretos.';
    if (lower.includes('email not confirmed')) return 'Seu e-mail ainda não foi confirmado. Abra a mensagem do Supabase no seu e-mail.';
    if (lower.includes('user already registered')) return 'Já existe uma conta com esse e-mail. Use “Entrar”.';
    if (lower.includes('password')) return 'A senha não atende aos requisitos. Use pelo menos 6 caracteres.';
    if (lower.includes('rate limit')) return 'Muitas tentativas em pouco tempo. Aguarde um pouco e tente novamente.';
    return text;
  }

  function setSyncStatus(text, status = '') {
    els.syncStatusText.textContent = text;
    els.syncStatus.className = `sync-pill${status ? ` ${status}` : ''}`;
    els.cloudStatusDetail.textContent = text;
  }
