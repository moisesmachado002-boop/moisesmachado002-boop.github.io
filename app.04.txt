id)
        .maybeSingle();
      if (error) throw error;
      const remote = data?.data ? normalizeState(data.data) : null;
      if (cloudDirty || (remote && stateTimestamp(state) > stateTimestamp(remote))) {
        await pushCloudState(true);
      } else if (remote && stateTimestamp(remote) > stateTimestamp(state)) {
        applyRemoteState(remote);
      } else if (!remote) {
        cloudDirty = true;
        await pushCloudState(true);
      } else {
        setSyncStatus('Sincronizado', 'synced');
      }
      if (showToast) toast('Sincronização concluída.');
    } catch (error) {
      console.error('Erro na sincronização:', error);
      setSyncStatus('Erro ao sincronizar', 'error');
      if (showToast) toast('Não foi possível sincronizar agora.');
    }
  }

  function applyRemoteState(remote) {
    const normalized = normalizeState(remote);
    if (stateTimestamp(normalized) && stateTimestamp(normalized) < stateTimestamp(state) && cloudDirty) return;
    state = normalized;
    cloudDirty = false;
    persistLocalState();
    populateOpeningBalancesForm();
    renderAll();
    setSyncStatus('Sincronizado', 'synced');
  }

  function subscribeRealtime() {
    if (!currentUser || !supabaseClient) return;
    if (realtimeChannel) supabaseClient.removeChannel(realtimeChannel);
    realtimeChannel = supabaseClient
      .channel(`finance-state-${currentUser.id}`)
      .on('postgres_changes', {
        event: '*', schema: 'public', table: 'finance_state', filter: `user_id=eq.${currentUser.id}`
      }, payload => {
        if ((payload.eventType === 'INSERT' || payload.eventType === 'UPDATE') && payload.new?.data) {
          const remote = normalizeState(payload.new.data);
          if (stateTimestamp(remote) > stateTimestamp(state)) applyRemoteState(remote);
          else setSyncStatus('Sincronizado', 'synced');
        }
      })
      .subscribe(status => {
        if (status === 'SUBSCRIBED' && !cloudDirty) setSyncStatus('Sincronizado', 'synced');
      });
  }

  function todayLocal() {
    const d = new Date();
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${y}-${m}-${day}`;
  }

  function currentMonth() {
    return todayLocal().slice(0, 7);
  }

  function setDefaultDates() {
    const month = currentMonth();
    [els.dashboardMonth, els.transactionMonthFilter, els.invoiceMonth, els.billsMonth, els.reportMonth].forEach(el => {

    });
    $$('input[type="date"]', document).forEach(el => { if (!el.value) el.value = todayLocal(); });
  }
