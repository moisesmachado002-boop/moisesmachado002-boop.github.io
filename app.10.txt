  async function importBackup(event) {
    const file = event.target.files?.[0]; if (!file) return;
    try {
      const parsed = JSON.parse(await file.text());
      const imported = {
        ...defaultState(),
        transactions: Array.isArray(parsed.transactions) ? parsed.transactions : [],
        cards: Array.isArray(parsed.cards) ? parsed.cards : [],
        purchases: Array.isArray(parsed.purchases) ? parsed.purchases : [],
        bills: Array.isArray(parsed.bills) ? parsed.bills : [],
        billPayments: Array.isArray(parsed.billPayments) ? parsed.billPayments : [],
        goals: Array.isArray(parsed.goals) ? parsed.goals : [],
        openingBalances: {
          account: parseMoney(parsed.openingBalances?.account || 0),
          cash: parseMoney(parsed.openingBalances?.cash || 0)
        }
      };
      if (!confirm('Importar este backup substituirá os dados atuais. Continuar?')) return;
      state = imported; saveState(); populateOpeningBalancesForm(); renderAll(); toast('Backup importado com sucesso.');
    } catch {
      toast('Não foi possível ler este backup.');
    } finally {
      event.target.value = '';
    }
  }

  function applySavedTheme() {
    const theme = localStorage.getItem(THEME_KEY) || (matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light');
    document.documentElement.dataset.theme = theme;
    $('#themeBtn').textContent = theme === 'dark' ? '☀' : '☾';
  }

  function toggleTheme() {
    const next = document.documentElement.dataset.theme === 'dark' ? 'light' : 'dark';
    document.documentElement.dataset.theme = next;
    localStorage.setItem(THEME_KEY, next);
    $('#themeBtn').textContent = next === 'dark' ? '☀' : '☾';
  }

  function parseMoney(value) {
    if (typeof value === 'number') return roundMoney(value);
    let s = String(value ?? '').trim().replace(/\s/g, '').replace(/R\$/gi, '');
    if (!s) return 0;
    if (s.includes(',') && s.includes('.')) s = s.replace(/\./g, '').replace(',', '.');
    else if (s.includes(',')) s = s.replace(',', '.');
    const n = Number(s.replace(/[^0-9.-]/g, ''));
    return Number.isFinite(n) ? roundMoney(n) : 0;
  }

  function money(value) {
    return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(Number(value) || 0);
  }

  function roundMoney(v) { return Math.round((Number(v) + Number.EPSILON) * 100) / 100; }
  function sum(arr, fn) { return arr.reduce((acc, item) => acc + Number(fn(item) || 0), 0); }
  function addMap(map, key, value) { map.set(key || 'Outros', roundMoney((map.get(key || 'Outros') || 0) + Number(value || 0))); }

  function localDate(dateStr) {
    const [y, m, d] = dateStr.split('-').map(Number);
    return new Date(y, m - 1, d, 12, 0, 0);
  }

  function formatDate(dateStr) {
    if (!dateStr) return '';
    return localDate(dateStr).toLocaleDateString('pt-BR');
  }

  function formatMonth(month) {
    const [y, m] = month.split('-').map(Number);
    return new Date(y, m - 1, 1).toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' });
  }

  function monthShort(month) {
    const [y, m] = month.split('-').map(Number);
    return new Date(y, m - 1, 1).toLocaleDateString('pt-BR', { month: 'short' }).replace('.', '');
  }
